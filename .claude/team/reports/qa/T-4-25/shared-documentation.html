// E3 shared transformation, version 1. A fixed artefact: do not edit per client.
// Ships as the E3 GTM custom template and as this JS helper.
function brPayload(br, rules) {
  var out = {};
  if (!br) return out;
  var props = br.properties || {};
  var item = br.item || {};
  var items = br.items || [];
  function defined(v) { return v !== undefined && v !== null; }
  Object.keys(props).forEach(function (k) { out[k] = props[k]; });
  Object.keys(item).forEach(function (k) { out[k] = item[k]; });
  (rules || []).forEach(function (r) {
    if (defined(props[r.name])) return; // a supplied value wins over the computed one
    if (r.sum) {
      var fields = r.sum.split('*').map(function (f) { return f.trim(); }).filter(Boolean);
      var total = 0;
      items.forEach(function (it) {
        var product = 1;
        fields.forEach(function (f) { product *= Number(it[f]) || 0; });
        total += product;
      });
      out[r.name] = Math.round(total * 1e6) / 1e6;
    } else if (r.form === 'ids') {
      out[r.name] = items.map(function (i) { return i[r.id_field]; }).filter(defined);
    } else {
      var q = r.quantity_field || 'quantity';
      out[r.name] = items.filter(function (i) { return defined(i[r.id_field]); }).map(function (i) {
        var o = {};
        o[r.id_field] = i[r.id_field];
        o.quantity = i[q]; // output key is always quantity
        return o;
      });
    }
  });
  return out;
}

// Ships as the E3 GTM custom template and as this JS helper.
function brPayload(br, rules) {
  var out = {};
  if (!br) return out;
  var props = br.properties || {};
  var item = br.item || {};
  var items = br.items || [];
  function defined(v) { return v !== undefined && v !== null; }
  Object.keys(props).forEach(function (k) { out[k] = props[k]; });
  Object.keys(item).forEach(function (k) { out[k] = item[k]; });
  (rules || []).forEach(function (r) {
    if (defined(props[r.name])) return; // a supplied value wins over the computed one
    if (r.sum) {
      var fields = r.sum.split('*').map(function (f) { return f.trim(); }).filter(Boolean);
      var total = 0;
      items.forEach(function (it) {
        var product = 1;
        fields.forEach(function (f) { product *= Number(it[f]) || 0; });
        total += product;
      });
      out[r.name] = Math.round(total * 1e6) / 1e6;
    } else if (r.form === 'ids') {
      out[r.name] = items.map(function (i) { return i[r.id_field]; }).filter(defined);
    } else {
      var q = r.quantity_field || 'quantity';
      out[r.name] = items.filter(function (i) { return defined(i[r.id_field]); }).map(function (i) {
        var o = {};
        o[r.id_field] = i[r.id_field];
        o.quantity = i[q]; // output key is always quantity
        return o;
      });
    }
  });
  return out;
}

// E3 shared transformation, version 1. A fixed artefact: do not edit per client.
// Ships as the E3 GTM custom template and as this JS helper.
function brPayload(br, rules) {
  var out = {};
  if (!br) return out;
  var props = br.properties || {};
  var item = br.item || {};
  var items = br.items || [];
  function defined(v) { return v !== undefined && v !== null; }
  Object.keys(props).forEach(function (k) { out[k] = props[k]; });
  Object.keys(item).forEach(function (k) { out[k] = item[k]; });
  (rules || []).forEach(function (r) {
    if (defined(props[r.name])) return; // a supplied value wins over the computed one
    if (r.sum) {
      var fields = r.sum.split('*').map(function (f) { return f.trim(); }).filter(Boolean);
      var total = 0;
      items.forEach(function (it) {
        var product = 1;
        fields.forEach(function (f) { product *= Number(it[f]) || 0; });
        total += product;
      });
      out[r.name] = Math.round(total * 1e6) / 1e6;
    } else if (r.form === 'ids') {
      out[r.name] = items.map(function (i) { return i[r.id_field]; }).filter(defined);
    } else {
      var q = r.quantity_field || 'quantity';
      out[r.name] = items.filter(function (i) { return defined(i[r.id_field]); }).map(function (i) {
        var o = {};
        o[r.id_field] = i[r.id_field];
        o.quantity = i[q]; // output key is always quantity
        return o;
      });
    }
  });
  return out;
}
